   ```typescript
   @Schema(...)
   class CostCalculation {
     // ... fields
     // Computed virtual:
   }
   CostCalculationSchema.virtual('totalCost').get(function () {
     return this.materials.reduce(...) + this.labor.reduce(...) * (1 + this.overheadPercent/100);
   });
   ```
